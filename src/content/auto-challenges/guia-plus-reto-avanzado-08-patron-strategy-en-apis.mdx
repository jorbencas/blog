### 💻 Código de la Solución (Python)

```python
class AuthStrategy:
    def autenticar(self, creds): pass

class BasicAuth(AuthStrategy):
    def autenticar(self, creds):
        return f"Basic {creds['user']}:{creds['pass']}"

class TokenAuth(AuthStrategy):
    def autenticar(self, creds):
        return f"Bearer {creds['token']}"

def sol(creds, estrategia):
    return estrategia.autenticar(creds)

creds_basic = {"user": "admin", "pass": "1234"}
creds_token = {"token": "eyJhbGciOiJIUzI1NiJ9"}
print(sol(creds_basic, BasicAuth()))
print(sol(creds_token, TokenAuth()))

```
