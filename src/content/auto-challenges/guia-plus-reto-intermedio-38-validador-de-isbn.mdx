- Identifica el propósito central del algoritmo

- Determina la estrategia óptima antes de escribir código

### 💻 Código de la Solución (JavaScript)

const sol = s => { s=s.replace(/-/g,''); if(s.length!==10) return false; let t=s.split('').reduce((a,c,i)=>a+(10-i)*(c==='X'?10:+c),0); return t%11===0; };
